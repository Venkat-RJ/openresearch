1 """
2 Gurobi implementation of the PDPTWL (Pickup and Delivery Problem with Time Windows
3 and LIFO Loading) compact three-index formulation.
4 
5 Based on: Cherkesly, Desaulniers, and Laporte (2015), Transportation Science.
6 Formulation: Constraints (1)-(13) from the paper.
7 
8 Assumptions (inferred):
9   - Travel costs c_{ij} = Euclidean distance (NOT SPECIFIED IN PAPER whether rounded).
10     We use the costs provided in the instance JSON directly.
11   - Travel times t_{ij} provided in the instance JSON directly.
12   - Vehicle fixed cost is included in c_{0,j} arcs (already in instance data).
13   - Big-M linearization used for constraints (8) and (10).
14 
15 Full LIFO is enforced by exact lazy separation of every incumbent route that
16 contains a pickup/pickup/delivery pattern inconsistent with stack order.
17 """

49 def lifo_lazy_callback(model, where):
50     """Separate a minimal infeasible path for every observed LIFO violation."""
51     if where != GRB.Callback.MIPSOL:
52         return
53     values = model.cbGetSolution(model._lifo_x)
54     for vehicle in model._lifo_vehicles:
55         successor = {
56             i: j
57             for (i, j) in model._lifo_arcs
58             if values[vehicle, i, j] > 0.5
59         }
60         sequence = [model._lifo_origin]
61         while sequence[-1] in successor and len(sequence) <= model._lifo_node_count:
62             sequence.append(successor[sequence[-1]])
63             if sequence[-1] == model._lifo_destination:
64                 break
65         stack = []
66         pickup_position = {}
67         for position, node in enumerate(sequence):
68             if 1 <= node <= model._lifo_request_count:
69                 stack.append(node)
70                 pickup_position[node] = position
71                 continue
72             if not (model._lifo_request_count < node <= 2 * model._lifo_request_count):
73                 continue
74             request = node - model._lifo_request_count
75             if request not in stack:
76                 continue
77             if stack[-1] != request:
78                 first = pickup_position[request]
79                 path = list(zip(sequence[first:position], sequence[first + 1:position + 1]))
80                 model.cbLazy(
81                     gp.quicksum(model._lifo_x[vehicle, i, j] for i, j in path)
82                     <= len(path) - 1
83                 )
84             stack.remove(request)

87 def build_model(data, time_limit):
88     """Build the PDPTWL compact formulation Gurobi model."""
89 
90     n = data["n_requests"]
91     n_nodes = data["n_nodes"]  # 2n + 2
92     Q = data["vehicle_capacity"]
93     planning_horizon = data["planning_horizon"]
94 
95     # Build node data
96     nodes = {}
97     for node_data in data["nodes"]:
98         nid = node_data["node_id"]
99         nodes[nid] = node_data
100 
101     P = list(range(1, n + 1))           # Pickup nodes
102     D = list(range(n + 1, 2 * n + 1))   # Delivery nodes
103     depot_o = 0
104     depot_d = 2 * n + 1
105     N = list(range(n_nodes))             # All nodes
106 
107     # Build arc set from instance data
108     arc_set = set()
109     arc_cost = {}
110     arc_time = {}
111     for arc in data["arcs"]:
112         i, j = arc["from"], arc["to"]
113         arc_set.add((i, j))
114         arc_cost[(i, j)] = arc["cost"]
115         arc_time[(i, j)] = arc["travel_time"]
116 
117     A = list(arc_set)
118 
119     # Determine number of vehicles K (upper bound: n vehicles)
120     # The paper says K is unrestricted; we use n as an upper bound
121     K_size = n
122     K = list(range(K_size))
123 
124     # Big-M values for linearization
125     M_time = planning_horizon[1] + max(
126         node["service_time"] for node in data["nodes"]
127     ) + max(arc["travel_time"] for arc in data["arcs"])
128     M_load = Q
129 
130     # --- Create model ---
131     model = gp.Model("PDPTWL")
132     model.setParam("Threads", 1)
133     model.setParam("TimeLimit", time_limit)
134     model.setParam("OutputFlag", 1)
135     model.setParam("LazyConstraints", 1)
136 
137     # --- Decision variables ---
138     # x[k,i,j] binary: vehicle k uses arc (i,j)
139     x = {}
140     for k in K:
141         for (i, j) in A:
142             x[k, i, j] = model.addVar(vtype=GRB.BINARY, name=f"x_{k}_{i}_{j}")
143 
144     # T[k,i] continuous: service start time at node i for vehicle k
145     T = {}
146     for k in K:
147         for i in N:
148             lb = nodes[i]["tw_early"]
149             ub = nodes[i]["tw_late"]
150             T[k, i] = model.addVar(lb=lb, ub=ub, vtype=GRB.CONTINUOUS,
151                                    name=f"T_{k}_{i}")
152 
153     # Q_var[k,i] continuous: load of vehicle k upon leaving node i
154     Q_var = {}
155     for k in K:
156         for i in N:
157             q_i = nodes[i]["load"]
158             lb = max(0, q_i)
159             ub = min(Q, Q + q_i)
160             Q_var[k, i] = model.addVar(lb=lb, ub=ub, vtype=GRB.CONTINUOUS,
161                                        name=f"Q_{k}_{i}")
162 
163     model.update()
164 
165     # --- Objective (1): minimize total cost ---
166     model.setObjective(
167         gp.quicksum(arc_cost[(i, j)] * x[k, i, j]
168                     for k in K for (i, j) in A),
169         GRB.MINIMIZE
170     )
171 
172     # --- Constraints ---
173 
174     # (2) Each pickup visited exactly once
175     for i in P:
176         outgoing = [(i, j) for (ii, j) in A if ii == i]
177         model.addConstr(
178             gp.quicksum(x[k, i, j] for k in K for (_, j) in outgoing) == 1,
179             name=f"visit_{i}"
180         )
181 
182     # (3) Pairing: if vehicle k visits pickup i, it must visit delivery n+i
183     for i in P:
184         ni = n + i  # delivery node for request i
185         for k in K:
186             out_pickup = [(i, j) for (ii, j) in A if ii == i]
187             out_delivery = [(ni, j) for (ii, j) in A if ii == ni]
188             model.addConstr(
189                 gp.quicksum(x[k, i, j] for (_, j) in out_pickup)
190                 - gp.quicksum(x[k, ni, j] for (_, j) in out_delivery) == 0,
191                 name=f"pair_{i}_{k}"
192             )
193 
194     # (4) Each vehicle leaves origin depot exactly once
195     out_depot = [(depot_o, j) for (ii, j) in A if ii == depot_o]
196     for k in K:
197         model.addConstr(
198             gp.quicksum(x[k, depot_o, j] for (_, j) in out_depot) == 1,
199             name=f"depot_out_{k}"
200         )
201 
202     # (5) Flow conservation at pickup and delivery nodes
203     for i in P + D:
204         in_arcs = [(ii, i) for (ii, jj) in A if jj == i]
205         out_arcs = [(i, j) for (ii, j) in A if ii == i]
206         for k in K:
207             model.addConstr(
208                 gp.quicksum(x[k, ii, i] for (ii, _) in in_arcs)
209                 - gp.quicksum(x[k, i, j] for (_, j) in out_arcs) == 0,
210                 name=f"flow_{i}_{k}"
211             )
212 
213     # (6) Each vehicle arrives at destination depot exactly once
214     in_depot = [(i, depot_d) for (i, jj) in A if jj == depot_d]
215     for k in K:
216         model.addConstr(
217             gp.quicksum(x[k, i, depot_d] for (i, _) in in_depot) == 1,
218             name=f"depot_in_{k}"
219         )
220 
221     # (8) Load constraints (linearized with big-M)
222     # Q_var[k,j] >= Q_var[k,i] + q_j - M*(1 - x[k,i,j])
223     for k in K:
224         for (i, j) in A:
225             q_j = nodes[j]["load"]
226             model.addConstr(
227                 Q_var[k, j] >= Q_var[k, i] + q_j - M_load * (1 - x[k, i, j]),
228                 name=f"load_{k}_{i}_{j}"
229             )
230 
231     # (9) Load bounds are set via variable bounds already
232 
233     # (10) Time constraints (linearized with big-M)
234     # T[k,j] >= T[k,i] + s_i + t_{ij} - M*(1 - x[k,i,j])
235     for k in K:
236         for (i, j) in A:
237             s_i = nodes[i]["service_time"]
238             t_ij = arc_time[(i, j)]
239             model.addConstr(
240                 T[k, j] >= T[k, i] + s_i + t_ij - M_time * (1 - x[k, i, j]),
241                 name=f"time_{k}_{i}_{j}"
242             )
243 
244     # (11) Time window bounds are set via variable bounds already
245 
246     # (12) Precedence: T[k,i] + s_i + t_{i,n+i} <= T[k,n+i]
247     for i in P:
248         ni = n + i
249         s_i = nodes[i]["service_time"]
250         if (i, ni) in arc_time:
251             t_i_ni = arc_time[(i, ni)]
252         else:
253             # Compute Euclidean distance if arc not in set
254             # INFERRED ASSUMPTION: direct travel time between pickup and delivery
255             xi, yi = nodes[i]["x"], nodes[i]["y"]
256             xni, yni = nodes[ni]["x"], nodes[ni]["y"]
257             t_i_ni = math.sqrt((xi - xni) ** 2 + (yi - yni) ** 2)
258         for k in K:
259             model.addConstr(
260                 T[k, i] + s_i + t_i_ni <= T[k, ni],
261                 name=f"prec_{i}_{k}"
262             )
263 
264     # --- LIFO constraints (7) ---
265     # We add LIFO constraints via infeasible path inequalities.
266     # For a compact formulation, we enumerate short LIFO-violating patterns.
267     # A LIFO violation occurs when pickup i is visited before pickup j,
268     # but delivery n+j is visited before delivery n+i (i.e., j is delivered
269     # before i, violating the "last picked up, first delivered" rule).
270     #
271     # For each pair (i, j) in P x P with i != j, the LIFO constraint says:
272     # If vehicle k picks up i then j (in that order), it must deliver j before i.
273     #
274     # We enforce: for each vehicle k, for each pair i,j in P:
275     #   x[k,i,j] = 1 implies n+j must be delivered before n+i
276     #   This is implicitly handled by the arc structure (no arcs from pickup i
277     #   to delivery n+j for j != i), but we add explicit LIFO ordering constraints.
278     #
279     # The arc set already excludes (i, n+j) for i in P, j in P, j != i,
280     # which handles direct pickup-to-wrong-delivery violations.
281     # For indirect violations, we add constraints for delivery ordering:
282     # If both i and j are onboard (picked up), then n+j must come before n+i
283     # if j was picked up after i. We model this with time-based constraints:
284     # T[k,n+j] <= T[k,n+i] + M*(2 - x[k,i,j] - (sum of x indicating j picked after i))
285     #
286     # INFERRED ASSUMPTION: The arc structure already restricts most LIFO violations.
287     # For the compact formulation, the paper notes that the explicit LIFO
288     # constraints (7) involve exponentially many subsets. We add pairwise
289     # LIFO constraints as a practical compromise for the compact model.
290     for i in P:
291         for j in P:
292             if i == j:
293                 continue
294             ni = n + i
295             nj = n + j
296             # If arc (i,j) exists (pickup i then pickup j), then delivery j
297             # must happen before delivery i: T[k,nj] <= T[k,ni]
298             if (i, j) in arc_set:
299                 for k in K:
300                     model.addConstr(
301                         T[k, nj] <= T[k, ni] + M_time * (1 - x[k, i, j]),
302                         name=f"lifo_{i}_{j}_{k}"
303                     )
304 
305     # --- Symmetry breaking ---
306     # Break symmetry between identical vehicles by ordering their first pickup
307     # INFERRED ASSUMPTION: Standard symmetry-breaking technique
308     for k in range(len(K) - 1):
309         # Vehicle k's departure time from depot <= vehicle k+1's departure time
310         model.addConstr(
311             T[k, depot_o] <= T[k + 1, depot_o],
312             name=f"sym_{k}"
313         )
314 
315     model.update()
316     model._lifo_x = gp.tupledict(x)
317     model._lifo_arcs = A
318     model._lifo_vehicles = K
319     model._lifo_origin = depot_o
320     model._lifo_destination = depot_d
321     model._lifo_request_count = n
322     model._lifo_node_count = len(N)
323     return model, x, T, Q_var, A, K, N, P, D, n, depot_o, depot_d, nodes

326 def extract_solution(model, x, T, Q_var, A, K, N, P, D, n, depot_o, depot_d, nodes):
327     """Extract solution from solved model."""
328     if model.SolCount == 0:
329         return None
330 
331     obj_val = model.ObjVal
332 
333     routes = []
334     for k in K:
335         route_arcs = []
336         for (i, j) in A:
337             if x[k, i, j].X > 0.5:
338                 route_arcs.append((i, j))
339 
340         if not route_arcs:
341             continue
342 
343         # Build route sequence from arcs
344         # Check if this is a non-empty route (not just depot->depot path)
345         visits_customer = any(i in P or i in D or j in P or j in D
346                               for (i, j) in route_arcs)
347         if not visits_customer:
348             continue
349 
350         # Build adjacency
351         adj = {}
352         for (i, j) in route_arcs:
353             adj[i] = j
354 
355         # Trace route from depot_o
356         route = [depot_o]
357         current = depot_o
358         visited_count = 0
359         while current in adj and visited_count < len(N):
360             nxt = adj[current]
361             route.append(nxt)
362             current = nxt
363             visited_count += 1
364             if current == depot_d:
365                 break
366 
367         # Get timing and load info
368         route_detail = []
369         for node_id in route:
370             detail = {
371                 "node_id": node_id,
372                 "arrival_time": round(T[k, node_id].X, 2),
373                 "load_after": round(Q_var[k, node_id].X, 2)
374             }
375             route_detail.append(detail)
376 
377         route_cost = sum(
378             model.getAttr("Obj", [x[k, i, j]])[0] * x[k, i, j].X
379             for (i, j) in route_arcs
380         )
381 
382         routes.append({
383             "vehicle_id": k,
384             "route_sequence": route,
385             "route_details": route_detail,
386         })
387 
388     solution = {
389         "objective_value": round(obj_val, 2),
390         "num_vehicles": len(routes),
391         "status": model.Status,
392         "status_description": {
393             GRB.OPTIMAL: "optimal",
394             GRB.TIME_LIMIT: "time_limit",
395             GRB.INFEASIBLE: "infeasible",
396             GRB.INF_OR_UNBD: "infeasible_or_unbounded",
397         }.get(model.Status, f"other_{model.Status}"),
398         "mip_gap": model.MIPGap if model.SolCount > 0 else None,
399         "best_bound": model.ObjBound if model.SolCount > 0 else None,
400         "lifo_enforcement": "exact_lazy_infeasible_path_separation",
401         "solve_time": model.Runtime,
402         "routes": routes,
403     }
404     return solution

407 def main():
408     parser = argparse.ArgumentParser(
409         description="PDPTWL Gurobi compact formulation solver"
410     )
411     parser.add_argument("--instance_path", type=str, required=True,
412                         help="Path to the instance JSON file")
413     parser.add_argument("--solution_path", type=str, required=True,
414                         help="Path for the output solution JSON file")
415     parser.add_argument("--time_limit", type=int, required=True,
416                         help="Maximum solver runtime in seconds")
417     parser.add_argument("--log_path", type=str, default=None, help="Path to log incumbent solutions")
418     args = parser.parse_args()
419     install_gurobi_logger(args.log_path)
420 
421     print(f"Loading instance from: {args.instance_path}")
422     data = load_instance(args.instance_path)
423 
424     print(f"Building model for {data['n_requests']} requests...")
425     model, x, T, Q_var, A, K, N, P, D, n, depot_o, depot_d, nodes = \
426         build_model(data, args.time_limit)
427 
428     print(f"Solving with time limit = {args.time_limit}s...")
429     model.optimize(lifo_lazy_callback)
430 
431     print(f"Status: {model.Status}, Solutions found: {model.SolCount}")
432 
433     if model.SolCount > 0:
434         solution = extract_solution(
435             model, x, T, Q_var, A, K, N, P, D, n, depot_o, depot_d, nodes
436         )
437         print(f"Objective value: {solution['objective_value']}")
438         print(f"Number of vehicles: {solution['num_vehicles']}")
439     else:
440         solution = {
441             "objective_value": None,
442             "num_vehicles": None,
443             "status": model.Status,
444             "status_description": "no_solution_found",
445             "mip_gap": None,
446             "best_bound": None,
447             "lifo_enforcement": "exact_lazy_infeasible_path_separation",
448             "solve_time": model.Runtime,
449             "routes": [],
450         }
451         print("No feasible solution found.")
452 
453     with open(args.solution_path, "w") as f:
454         solution["runtime"] = time.time() - _GUROBI_CODE_START_TIME
455         json.dump(solution, f, indent=2)
456     print(f"Solution saved to: {args.solution_path}")
457

459 if __name__ == "__main__":
